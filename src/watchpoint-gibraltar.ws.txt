rule ("7.1: Player Joins") {
    event {
        Player Joined Match;
        All;
        All;
    }
    conditions {
        Is Dummy Bot(Event Player) == False;
    }
    actions {
        Set Player Variable(Event Player, isLoading, True);
        If(Not(Array Contains(Global.players, Custom String("{0}", Event Player))));
            If(Compare(Event Player, ==, Host Player));
                Set Global Variable(hostPlayerName, Custom String("{0}", Event Player));
            End;
            Modify Global Variable At Index(lobbyStats, 0, Add, 1);
            Set Global Variable At Index(players, Count Of(Global.players), Custom String("{0}", Event Player));
        Else;
            Modify Global Variable At Index(lobbyStats, 1, Add, 1);
        End;
        Wait Until(Has Spawned(Event Player), 99999999);
        Disable Game Mode HUD(Event Player);
        Disable Kill Feed(Event Player);
        Disable Game Mode In-World UI(Event Player);
        Disable Nameplates(Event Player, All Players(All Teams));
        Disable Nameplates(All Players(All Teams), Event Player);
        Disable Movement Collision With Players(Event Player);
        Wait(0.016, Ignore Condition);
        Call Subroutine(loadLobby);
        Wait(0.1, Ignore Condition);
        Set Player Variable(Event Player, isLoading, False);
    }
}

rule ("14.3: Update Run Timer Format") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        (Event Player).isTimerActive == False;
        (Event Player).isRunPaused == False;
        (Event Player).isRunLoaded == True;
        First Of(Global.settings_) == False;
    }
    actions {
        Modify Global Variable(playersUnpaused, Add, 1);
        Set Player Variable(Event Player, isTimerActive, True);
        While(Not((Event Player).isRunPaused));
            If(Compare(Global.playersUnpaused, >=, 10));
                Wait(Add(0.096, Multiply(Server Load, 0.0004)), Ignore Condition);
            Else If(Compare(Global.playersUnpaused, >=, 8));
                Wait(Add(0.08, Multiply(Server Load, 0.0004)), Ignore Condition);
            Else If(Compare(Global.playersUnpaused, >=, 6));
                Wait(Add(0.064, Multiply(Server Load, 0.0004)), Ignore Condition);
            Else If(Compare(Global.playersUnpaused, >=, 4));
                Wait(Add(0.048, Multiply(Server Load, 0.0006)), Ignore Condition);
            Else If(Compare(Global.playersUnpaused, >=, 2));
                Wait(Add(0.032, Multiply(Server Load, 0.0008)), Ignore Condition);
            Else;
                Wait(Add(0.016, Multiply(Server Load, 0.001)), Ignore Condition);
            End;
            Set Player Variable(Event Player, runTimeFormatted, If-Then-Else(Compare((Event Player).runTime, >=, 3600), Custom String("{0}:{1}:{2}", Round To Integer(Divide((Event Player).runTime, 3600), Down), String Slice(Add(Divide(Modulo((Event Player).runTime, 3600), 60), 100), 1, 2), String Slice(Add(Modulo((Event Player).runTime, 60), 100), 1, 9999)), If-Then-Else(Compare((Event Player).runTime, >=, 60), Custom String("{0}:{1}", Round To Integer(Divide((Event Player).runTime, 60), Down), String Slice(Add(Modulo((Event Player).runTime, 60), 100), 1, 9999)), Custom String("{0}.{1}", Round To Integer((Event Player).runTime, Down), String Slice(Add(Round To Integer(Multiply(Modulo((Event Player).runTime, 1), 100), To Nearest), 100), 1, 2)))));
        End;
        Modify Global Variable(playersUnpaused, Subtract, 1);
        Set Player Variable(Event Player, isTimerActive, False);
    }
}

    actions {
        Set Player Variable(Event Player, isLoading, True);
        Set Player Variable(Event Player, isRunLegit, False);
        If(Not((Event Player).isRunPaused));
            Stop Chasing Player Variable(Event Player, runTime);
            Set Player Variable(Event Player, isRunPaused, True);
            If(Compare(Distance Between(Position Of(Event Player), First Of((Event Player).cpData)), >=, Value In Array((Event Player).cpData, 1)));
                Modify Player Variable(Event Player, runMistakes, Add, 1);
            End;
        End;
        Modify Player Variable(Event Player, runCurrentCp, Add, 1);
        Call Subroutine(loadCp);
        Call Subroutine(cpResetPlayer);
        Destroy In-World Text((Event Player).nameplateObject);
        Call Subroutine(loadNameplate);
        Set Facing(Event Player, Direction Towards(Eye Position(Event Player), Value In Array((Event Player).cpData, 6)), To World);
        Wait Until(Not(And(Is Button Held(Event Player, Button(Ultimate)), Is Button Held(Event Player, Button(Secondary Fire)))), 0.25);
        If(And(And(Is Button Held(Event Player, Button(Ultimate)), Is Button Held(Event Player, Button(Secondary Fire))), Compare((Event Player).runCurrentCp, <, Subtract(Count Of(Global.cps), 1))));
            Set Player Variable(Event Player, isLoading, True);
            While(And(And(Is Button Held(Event Player, Button(Ultimate)), Is Button Held(Event Player, Button(Secondary Fire))), Compare((Event Player).runCurrentCp, <, Subtract(Count Of(Global.cps), 1))));
                Modify Player Variable(Event Player, runCurrentCp, Add, 1);
                Teleport(Event Player, First Of(Value In Array(Global.cps, (Event Player).runCurrentCp)));
                Wait(0.02, Ignore Condition);
            End;
            Call Subroutine(loadCp);
            Call Subroutine(cpResetPlayer);
            Destroy In-World Text((Event Player).nameplateObject);
            Call Subroutine(loadNameplate);
            Set Facing(Event Player, Direction Towards(Eye Position(Event Player), Value In Array((Event Player).cpData, 6)), To World);
        End;
        If(Compare((Event Player).runCurrentCp, ==, Subtract(Count Of(Global.cps), 1)));
            Call Subroutine(finalCpReached);
    }
}

rule ("21.7.1: Freemode Input") {
    event {
        Ongoing - Each Player;
        All;
        All;
    }
    conditions {
        Is Button Held(Event Player, Button(Ultimate)) == True;
        Is Button Held(Event Player, Button(Primary Fire)) == False;
        Is Button Held(Event Player, Button(Secondary Fire)) == False;
        Is Button Held(Event Player, Button(Reload)) == False;
        Is Button Held(Event Player, Button(Interact)) == True;
        Is Button Held(Event Player, Button(Ability 2)) == False;
        (Event Player).isInLobby == False;
        (Event Player).runCurrentCp < Subtract(Count Of(Global.cps), 1);
    }
    actions {
        If(Not((Event Player).isInFreemode));
            Stop Chasing Player Variable(Event Player, runTime);
            Set Player Variable(Event Player, isRunPaused, True);
            Set Player Variable(Event Player, isRunLegit, False);
            If(Compare(Distance Between(Position Of(Event Player), First Of((Event Player).cpData)), >=, Value In Array((Event Player).cpData, 1)));
                Modify Player Variable(Event Player, runMistakes, Add, 1);
            End;
            Set Secondary Fire Enabled(Event Player, True);
            Set Player Variable(Event Player, isInFreemode, True);
            Call Subroutine(unloadFreemodeHud);
            Call Subroutine(loadFreemodeHud);
        Else;
            Set Player Variable(Event Player, isLoading, True);
            Set Player Variable(Event Player, isInFreemode, False);
            Call Subroutine(unloadFreemodeHud);
            If(Compare(Distance Between(Position Of(Event Player), First Of((Event Player).cpData)), >=, Value In Array((Event Player).cpData, 1)));
                Call Subroutine(cpResetPlayer);
            End;
            Set Secondary Fire Enabled(Event Player, Value In Array((Event Player).cpData, 2));
            If(Not(Value In Array((Event Player).cpData, 3)));
                Disallow Button(Event Player, Button(Crouch));
            Else;
                Allow Button(Event Player, Button(Crouch));
            End;
            Set Player Variable(Event Player, isLoading, False);
        End;
        Destroy In-World Text((Event Player).nameplateObject);
        Call Subroutine(loadNameplate);
    }
}
